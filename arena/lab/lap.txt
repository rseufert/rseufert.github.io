; LAP: not on the roster. The idea that aimed right and fired too slowly.
; Everything is worked out on the field, not from the turret: each pass
; the scan is turned into where the target is (a sine table, our position,
; the bearing and range), so our own driving never spoils the sums. A
; runner is shot ahead of itself. A tank going round in tight circles
; (CIRCLER, once its steer queue saturates: 45 ticks a lap, 40 across)
; comes back to where it was, so for it the shots are paced to a lap less
; the flight and the latency, and each is aimed where it was a shot ago,
; which is where it will be when the missile lands. That part works: within
; two degrees, at a shot every 28 ticks. What does not: the pass is too
; long to keep up with the target between shots, the circling test takes
; too long to trip, and the whole is outscored by HERON's plain orbit. To
; finish it, sample the target every eight ticks or so in a loop that does
; nothing else, and leave the sums to the gaps. See arena/lab/README.md.
; memory: [21] tick last pass  [23] range  [24] which way round  [25] at a wall
; [26] next scan's tick (lap)  [37] scan tick (lap)  [27] aim, absolute  [30] passes  [31] ticks of last pass
; [32] refinds  [33] laps entered  [34] now  [35] lap shots  [36] empty sweeps
; [40..47],[48..55],[56..63] ring of 8 by pass: x, y, tick  [70],[71] us  [72],[73] sin, cos
; [74],[75] it now  [76],[77] its speed x16  [78..81] it a shot ago, two shots ago (lap)
; [82],[83] the point to shoot at  [84],[85] the point to look at next
; [100..190] sine x1000 by degree

:lapmode

  jmp lap
:lscan

; ---- where it is: from our place, the bearing and the range ----
:where
  int @locate
  mov [70], ax
  mov [71], bx
  ipo theading, ax
  call sine
  mov [72], cx
  ipo theading, ax
  add ax, 90
  call sine
  mov [73], cx

  div ax, 1000
  add ax, [70]
  mov [74], ax
  mov ax, [23]
  mpy ax, [72]
  div ax, 1000
  add ax, [71]
  mov [75], ax

:sine                   ; ax = degrees -> cx = sine x1000
  add ax, 720
  mod ax, 360
  mov dx, 1
  cmp ax, 180
  jle s1
  sub ax, 180
  mov dx, -1
:s1
  cmp ax, 90
  jle s2
  neg ax
  add ax, 180
:s2

  mov cx, [ax]
  mpy cx, dx
  ret
; ---- driving: off the walls, else across its line of sight at 380-550 ----
:drive

  jl wall

  jg cap2
  mov bx, -40
:cap2

  opo throttle, 100
